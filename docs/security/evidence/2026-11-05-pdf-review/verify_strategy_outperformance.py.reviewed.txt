def classify_all_long_archetypes(features: LongArchetypeFeatures) -> list[SetupLabel]:
    """Return ALL matched long archetypes in priority order instead of stopping at the first."""
    matched: list[SetupLabel] = []
    atr = max(features.atr14_4h, Decimal("1e-8"))
    funding_8h = features.funding_rate_8h
    oi_7d = features.oi_delta_7d or Decimal("0")
    oi_4h = features.oi_delta_4h or Decimal("0")
    taker_4h = features.taker_buy_ratio_4h

    # 0a. FUNDING_SQUEEZE_SPRING_LONG
    is_funding_fuel = (
        funding_8h is not None
        and funding_8h <= Decimal("-0.00015")
        and features.oi_delta_24h is not None
        and features.oi_delta_24h >= Decimal("0.06")
        and (
            (features.taker_buy_ratio_1h is not None and features.taker_buy_ratio_1h >= Decimal("1.18"))
            or (taker_4h is not None and taker_4h >= Decimal("1.18"))
        )
    )
    low_4h = features.low_4h or features.low_24h
    holds_higher_low = (features.low_30d > Decimal("0")) and (low_4h >= features.low_30d * Decimal("1.03"))
    inside_value_area = (features.close >= features.val_30d) and (features.pos_in_range_30d >= Decimal("0.15"))
    ema20_4h_val = features.ema20_4h or features.ema50_4h
    reclaims_emas = (
        features.close >= ema20_4h_val
        and features.close >= features.ema50_4h * Decimal("0.965")
        and (
            (features.rvol_1h is not None and features.rvol_1h >= Decimal("1.35"))
            or features.rvol_4h >= Decimal("1.35")
        )
    )
    if is_funding_fuel and holds_higher_low and inside_value_area and reclaims_emas:
        matched.append(SetupLabel.FUNDING_SQUEEZE_SPRING_LONG)

    # 0b. SHORT_SQUEEZE_ABSORPTION_LONG
    is_funding_crowded_short = funding_8h is not None and funding_8h <= Decimal("-0.00005")
    is_oi_expanding = oi_7d >= Decimal("0.15") or oi_4h >= Decimal("0.035")
    has_buyer_absorption = features.cvd_24h > Decimal("0") or (taker_4h is not None and taker_4h >= Decimal("1.00"))
    holding_structure_0b = (
        features.close >= features.ema50_4h * Decimal("0.975")
        or features.close >= features.poc_30d * Decimal("0.975")
    )
    if is_funding_crowded_short and is_oi_expanding and has_buyer_absorption and holding_structure_0b:
        matched.append(SetupLabel.SHORT_SQUEEZE_ABSORPTION_LONG)

    # 1. COIL_SPRING_LONG
    spring_threshold = min(features.ema50_4h, features.poc_30d) - Decimal("0.10") * atr
    reclaim_threshold = max(features.ema50_4h, features.poc_30d * Decimal("0.99"))
    if (
        features.low_24h < spring_threshold
        and features.close >= reclaim_threshold
        and features.vcr_4h <= Decimal("0.90")
    ):
        matched.append(SetupLabel.COIL_SPRING_LONG)

    is_mom_up = (features.macd_hist_4h >= features.macd_hist_4h_prev) or (features.macd_hist_4h > Decimal("0"))

    # 2. VCP_COMPRESSION_LONG
    is_vcp_compressed = features.bb_inside_kc_4h or (features.bb_bandwidth_4h <= Decimal("0.18"))
    has_supply_exhaustion = (features.vcr_4h <= Decimal("1.10")) or (features.rvol_4h <= Decimal("1.15"))
    if (
        features.range_height_30d <= Decimal("0.45")
        and is_vcp_compressed
        and has_supply_exhaustion
        and is_mom_up
        and features.close >= features.ema50_4h * Decimal("0.980")
        and features.dist_to_ceiling <= Decimal("0.12")
        and features.pos_in_range_30d > Decimal("0.60")
    ):
        matched.append(SetupLabel.VCP_COMPRESSION_LONG)

    # 3. OI_ACCUMULATION_LONG
    is_compressed = features.bb_inside_kc_4h or (features.bb_bandwidth_4h <= Decimal("0.22"))
    holding_structure = (
        (
            features.close >= features.ema50_4h * Decimal("0.975")
            or features.close >= features.poc_30d * Decimal("0.975")
        )
        and (features.val_30d <= Decimal("0") or features.close >= features.val_30d)
    )
    is_stealth_compressed = (
        Decimal("-0.04") <= features.ret_24h <= Decimal("0.09")
        and features.bb_bandwidth_4h <= Decimal("0.20")
    )
    has_acute_derivative_loading = (
        (
            (features.oi_delta_4h is not None and features.oi_delta_4h >= Decimal("0.05"))
            or (features.oi_delta_24h is not None and features.oi_delta_24h >= Decimal("0.12"))
        )
        and (
            (features.taker_buy_ratio_1h is not None and features.taker_buy_ratio_1h >= Decimal("1.15"))
            or (taker_4h is not None and taker_4h >= Decimal("1.15"))
        )
    )
    if (is_stealth_compressed and has_acute_derivative_loading and holding_structure) or (
        Decimal("0.08") <= oi_7d <= Decimal("0.60")
        and is_compressed
        and abs(features.ret_7d) <= Decimal("0.12")
        and is_mom_up
        and holding_structure
    ):
        matched.append(SetupLabel.OI_ACCUMULATION_LONG)

    # 4. RANGE_BOTTOM_ACCUMULATION_LONG
    if (
        features.close >= features.val_30d
        and features.range_height_30d <= Decimal("0.55")
        and (is_compressed or features.bb_bandwidth_4h <= Decimal("0.24"))
        and is_mom_up
        and features.close >= features.ema50_4h * Decimal("0.975")
        and features.pos_in_range_30d <= Decimal("0.60")
        and features.vcr_4h <= Decimal("1.20")
        and taker_4h is not None and taker_4h >= Decimal("0.95")
    ):
        matched.append(SetupLabel.RANGE_BOTTOM_ACCUMULATION_LONG)

    # 5. BREAKOUT_LONG
    at_or_above_resistance = (features.close >= features.local_swing_high_4h) or (
        features.close >= features.vah_30d
    )
    has_breakout_volume = (
        (features.rvol_4h >= Decimal("1.25") and features.oi_delta_4h is not None and oi_4h >= Decimal("0.0"))
        or (features.rvol_4h >= Decimal("2.0"))
    )
    if at_or_above_resistance and has_breakout_volume and features.norm_momentum_1h > Decimal("0"):
        matched.append(SetupLabel.BREAKOUT_LONG)

    # 6. MOMENTUM_EMERGING_LONG
    if (
        features.btc_residual_alpha_24h >= Decimal("0.03")
        and features.macd_hist_4h > features.macd_hist_4h_prev
        and features.macd_hist_1h > Decimal("0")
        and taker_4h is not None and taker_4h >= Decimal("1.15")
    ):
        matched.append(SetupLabel.MOMENTUM_EMERGING_LONG)

    # 7. Fallbacks
    if at_or_above_resistance and has_breakout_volume and SetupLabel.BREAKOUT_LONG not in matched:
        matched.append(SetupLabel.BREAKOUT_LONG)
    if oi_7d >= Decimal("0.05") and is_compressed and is_mom_up and holding_structure and SetupLabel.OI_ACCUMULATION_LONG not in matched:
        matched.append(SetupLabel.OI_ACCUMULATION_LONG)
    if is_compressed or features.bb_bandwidth_4h <= Decimal("0.22"):
        if features.pos_in_range_30d > Decimal("0.60") and has_supply_exhaustion and is_mom_up and SetupLabel.VCP_COMPRESSION_LONG not in matched:
            matched.append(SetupLabel.VCP_COMPRESSION_LONG)
        is_val_support_holding = (
            features.close >= features.val_30d
            and features.close >= features.ema50_4h * Decimal("0.975")
        )
        if (
            features.pos_in_range_30d <= Decimal("0.60")
            and is_val_support_holding
            and is_mom_up
            and taker_4h is not None and taker_4h >= Decimal("0.95")
            and SetupLabel.RANGE_BOTTOM_ACCUMULATION_LONG not in matched
        ):
            matched.append(SetupLabel.RANGE_BOTTOM_ACCUMULATION_LONG)
        if (
            features.bb_bandwidth_4h <= Decimal("0.15")
            and is_val_support_holding
            and features.close >= features.poc_30d * Decimal("0.985")
            and taker_4h is not None and taker_4h >= Decimal("0.95")
            and has_supply_exhaustion
        ):
            lbl = SetupLabel.VCP_COMPRESSION_LONG if features.pos_in_range_30d > Decimal("0.55") else SetupLabel.RANGE_BOTTOM_ACCUMULATION_LONG
            if lbl not in matched:
                matched.append(lbl)
    return matched


def evaluate_522_population_simulations() -> dict[str, Any]:
    """
    Loads the full 522-symbol 1h klines (62,640 bars) and 1h OI (62,640 points) from
    docs/security/evidence/2026-11-05-universe-hourly-*.json and 2026-11-05-universe-oi-*.json.
    Evaluates:
      1. Pre-Breakout Coiled Entry Core (4H/1H Squeeze + Support + Volume/Flow)
      2. Track 3 Watch Archetype W1 (STEALTH_FLOW_DIVERGENCE_WATCH)
      3. Track 3 Watch Archetype W2 (DEEP_VALUE_RECLAIM_WATCH)
      4. Track 3 Watch Archetype W3 (IMPULSE_CONTINUATION_FLAG_WATCH)
    at two pre-event cutoffs:
      - Window A: Nov 3 23:00Z -> predicting Nov 4 calendar-day top-10 gainers & 24h returns across all 512 eligible symbols
      - Window B: Nov 4 09:00Z -> predicting Nov 5 rolling-24h top-10 gainers & 24h returns across all 512 eligible symbols
    """
    hourly_files = sorted(EVIDENCE_DIR.glob("2026-11-05-universe-hourly-*.json"))
    oi_files = sorted(EVIDENCE_DIR.glob("2026-11-05-universe-oi-*.json"))

    klines_by_sym: dict[str, list[dict[str, Any]]] = {}
    for hf in hourly_files:
        data = json.loads(hf.read_text())
        for sym, obj in data.get("results", {}).items():
            raw_rows = obj.get("result", []) if isinstance(obj, dict) else []
            parsed = []
            for r in raw_rows:
                if isinstance(r, list) and len(r) >= 11:
                    parsed.append({
                        "open_time": int(r[0]),
                        "open": float(r[1]),
                        "high": float(r[2]),
                        "low": float(r[3]),
                        "close": float(r[4]),
                        "volume": float(r[5]),
                        "close_time": int(r[6]),
                        "quote_volume": float(r[7]),
                        "taker_buy_base": float(r[9]),
                        "taker_buy_quote": float(r[10]),
                    })
            if parsed:
                klines_by_sym[sym] = parsed

    oi_by_sym: dict[str, list[dict[str, Any]]] = {}
    for of in oi_files:
        data = json.loads(of.read_text())
        for sym, obj in data.get("results", {}).items():
            raw_rows = obj.get("result", []) if isinstance(obj, dict) else []
            if isinstance(raw_rows, list) and raw_rows:
                oi_by_sym[sym] = raw_rows

    bottleneck_verified = json.loads((EVIDENCE_DIR / "2026-11-05-gainers-bottleneck-analysis-verified.json").read_text())
    nov4_top10 = [x["symbol"] for x in bottleneck_verified["yesterday_rome_top10"]]
    nov5_top10 = [x["symbol"] for x in bottleneck_verified["current_observed_top10"]]

    print(f"Loaded {len(klines_by_sym)} symbols of 1h klines and {len(oi_by_sym)} symbols of 1h OI.")
    print(f"Nov 4 Top 10: {nov4_top10}")
    print(f"Nov 5 Top 10: {nov5_top10}")

    # Helper to compute EMA, Bollinger, ATR on 1h / resampled 4h
    def calc_ema(vals: list[float], period: int) -> float:
        if not vals:
            return 0.0
        k = 2.0 / (period + 1.0)
        e = vals[0]
        for v in vals[1:]:
            e = v * k + e * (1.0 - k)
        return e

    def evaluate_window(cutoff_ms: int, horizon_end_ms: int, top10_set: set[str], label: str) -> dict[str, Any]:
        # First compute forward 24h return for every symbol with complete data
        forward_ret: dict[str, float] = {}
        forward_mfe: dict[str, float] = {}
        forward_mae: dict[str, float] = {}
        features_by_sym: dict[str, dict[str, Any]] = {}

        for sym, bars in klines_by_sym.items():
            past = [b for b in bars if b["close_time"] <= cutoff_ms]
            future = [b for b in bars if cutoff_ms < b["close_time"] <= horizon_end_ms]
            if len(past) < 72 or len(future) < 20:
                continue

            anchor_close = float(past[-1]["close"])
            if anchor_close <= 0:
                continue

            # 24hquote vol at cutoff
            qvol_24h = sum(float(b["quote_volume"]) for b in past[-24:])
            if qvol_24h < 1_500_000:  # standard Atlas $1.5M liquidity hygiene floor
                continue

            end_close = float(future[-1]["close"])
            max_high = max(float(b["high"]) for b in future)
            min_low = min(float(b["low"]) for b in future)

            f_ret = (end_close - anchor_close) / anchor_close
            f_mfe = (max_high - anchor_close) / anchor_close
            f_mae = (min_low - anchor_close) / anchor_close
            forward_ret[sym] = f_ret
            forward_mfe[sym] = f_mfe
            forward_mae[sym] = f_mae

            # Compute 1h and 4h features at cutoff
            closes_1h = [float(b["close"]) for b in past]
            highs_1h = [float(b["high"]) for b in past]
            lows_1h = [float(b["low"]) for b in past]
            qvols_1h = [float(b["quote_volume"]) for b in past]
            vols_1h = [float(b["volume"]) for b in past]
            taker_buy_vols_1h = [float(b.get("taker_buy_base", b.get("taker_buy_volume", 0.0))) for b in past]

            # Build 4h bars from 1h bars
            bars_4h = []
            for i in range(0, len(past) - 3, 4):
                chunk = past[i : i + 4]
                bars_4h.append({
                    "open": float(chunk[0]["open"]),
                    "high": max(float(x["high"]) for x in chunk),
                    "low": min(float(x["low"]) for x in chunk),
                    "close": float(chunk[-1]["close"]),
                    "qvol": sum(float(x["quote_volume"]) for x in chunk),
                })
            closes_4h = [b["close"] for b in bars_4h]
            ema20_4h = calc_ema(closes_4h, 20)
            ema50_4h = calc_ema(closes_4h, 50)
            ema20_1h = calc_ema(closes_1h, 20)
            ema50_1h = calc_ema(closes_1h, 50)

            # 4h Bollinger Bandwidth (20-period)
            if len(closes_4h) >= 20:
                w20 = closes_4h[-20:]
                mean20 = sum(w20) / 20.0
                std20 = (sum((x - mean20) ** 2 for x in w20) / 20.0) ** 0.5
                bbw_4h = (4.0 * std20) / max(mean20, 1e-8)
            else:
                bbw_4h = 0.25

            # 1h Bollinger Bandwidth (12-period and 20-period)
            w20_1h = closes_1h[-20:]
            mean20_1h = sum(w20_1h) / 20.0
            std20_1h = (sum((x - mean20_1h) ** 2 for x in w20_1h) / 20.0) ** 0.5
            bbw_1h = (4.0 * std20_1h) / max(mean20_1h, 1e-8)

            # Range position over available window (up to 96h)
            high_win = max(highs_1h)
            low_win = min(lows_1h)
            pos_in_range = (anchor_close - low_win) / max(high_win - low_win, 1e-8)

            # Volume ratios
            vol_6h = sum(qvols_1h[-6:]) / 6.0
            vol_prior_18h = sum(qvols_1h[-24:-6]) / 18.0
            vol_ratio_6h = vol_6h / max(vol_prior_18h, 1e-8)

            rvol_1h = qvols_1h[-1] / max(sum(qvols_1h[-21:-1]) / 20.0, 1e-8)
            rvol_4h = sum(qvols_1h[-4:]) / max(sum(qvols_1h[-24:-4]) / 5.0, 1e-8)

            # Taker buy ratio 4h & 1h
            tot_v_4h = sum(vols_1h[-4:])
            tb_v_4h = sum(taker_buy_vols_1h[-4:])
            sell_v_4h = max(tot_v_4h - tb_v_4h, 1e-8)
            taker_ratio_4h = tb_v_4h / sell_v_4h if tot_v_4h > 0 else 1.0

            # Returns
            ret_6h = (anchor_close - closes_1h[-7]) / max(closes_1h[-7], 1e-8)
            ret_24h = (anchor_close - closes_1h[-25]) / max(closes_1h[-25], 1e-8)
            ret_72h = (anchor_close - closes_1h[-72]) / max(closes_1h[-72], 1e-8)

            # OI changes
            oi_rows = [r for r in oi_by_sym.get(sym, []) if r["timestamp"] <= cutoff_ms]
            oi_4h = 0.0
            oi_6h = 0.0
            oi_24h = 0.0
            if len(oi_rows) >= 25:
                oi_now = float(oi_rows[-1]["sumOpenInterest"])
                oi_p4 = float(oi_rows[-5]["sumOpenInterest"])
                oi_p6 = float(oi_rows[-7]["sumOpenInterest"])
                oi_p24 = float(oi_rows[-25]["sumOpenInterest"])
                oi_4h = (oi_now - oi_p4) / max(oi_p4, 1e-8) if oi_p4 > 0 else 0.0
                oi_6h = (oi_now - oi_p6) / max(oi_p6, 1e-8) if oi_p6 > 0 else 0.0
                oi_24h = (oi_now - oi_p24) / max(oi_p24, 1e-8) if oi_p24 > 0 else 0.0

            features_by_sym[sym] = {
                "close": anchor_close,
                "qvol_24h": qvol_24h,
                "ema20_4h": ema20_4h,
                "ema50_4h": ema50_4h,
                "ema20_1h": ema20_1h,
                "ema50_1h": ema50_1h,
                "bbw_4h": bbw_4h,
                "bbw_1h": bbw_1h,
                "pos_in_range": pos_in_range,
                "vol_ratio_6h": vol_ratio_6h,
                "rvol_1h": rvol_1h,
                "rvol_4h": rvol_4h,
                "taker_ratio_4h": taker_ratio_4h,
                "ret_6h": ret_6h,
                "ret_24h": ret_24h,
                "ret_72h": ret_72h,
                "oi_4h": oi_4h,
                "oi_6h": oi_6h,
                "oi_24h": oi_24h,
            }

        universe_syms = list(features_by_sym.keys())
        u_mean_ret = sum(forward_ret[s] for s in universe_syms) / len(universe_syms)
        u_mean_mfe = sum(forward_mfe[s] for s in universe_syms) / len(universe_syms)
        u_mean_mae = sum(forward_mae[s] for s in universe_syms) / len(universe_syms)

        # Define Strategy Selectors (each capped at top-12 by its ranking score for matched-budget comparison)
        # 1. Current Pre-Breakout Coiled Core Proxy (Strict compression bbw_4h <= 0.22, above EMA50_4h, non-chasing ret_24h in [-8%, +14%], positive flow)
        coiled_core_candidates = []
        for s, f in features_by_sym.items():
            if (
                f["close"] >= f["ema50_4h"] * 0.985
                and f["bbw_4h"] <= 0.22
                and -0.08 <= f["ret_24h"] <= 0.14
                and f["pos_in_range"] >= 0.25
                and (f["rvol_4h"] >= 1.15 or f["oi_24h"] >= 0.05)
                and f["oi_4h"] >= -0.025
            ):
                # Rank by compression tightness + OI + taker/vol
                score = (0.25 - f["bbw_4h"]) * 100 + f["oi_24h"] * 80 + f["vol_ratio_6h"] * 10
                coiled_core_candidates.append((score, s))
        coiled_core_candidates.sort(reverse=True)

        # 2. Challenger Pre-Breakout Coiled Entry (With Stage 1 OI/Absorption unlocked: prioritizes coiled setups with true 4h/24h OI buildup + pocket pivot)
        challenger_core_candidates = []
        for s, f in features_by_sym.items():
            if (
                f["close"] >= f["ema50_4h"] * 0.975
                and f["bbw_4h"] <= 0.24
                and -0.08 <= f["ret_24h"] <= 0.16
                and f["pos_in_range"] >= 0.18
                and (f["oi_6h"] >= 0.03 or f["oi_24h"] >= 0.06 or f["vol_ratio_6h"] >= 1.35)
                and f["oi_4h"] >= -0.02
            ):
                score = f["oi_6h"] * 120 + f["oi_24h"] * 80 + f["vol_ratio_6h"] * 15 + (0.25 - f["bbw_4h"]) * 60
                challenger_core_candidates.append((score, s))
        challenger_core_candidates.sort(reverse=True)

        # 3. Track 3 Watch W1: STEALTH_FLOW_DIVERGENCE_WATCH (Volume + OI surge while price still quiet)
        w1_candidates = []
        for s, f in features_by_sym.items():
            if (
                f["vol_ratio_6h"] >= 1.50
                and (f["oi_6h"] >= 0.04 or f["oi_24h"] >= 0.08)
                and -0.06 <= f["ret_6h"] <= 0.10
            ):
                score = f["oi_6h"] * 150 + f["vol_ratio_6h"] * 25
                w1_candidates.append((score, s))
        w1_candidates.sort(reverse=True)

        # 4. Track 3 Watch W2: DEEP_VALUE_RECLAIM_WATCH (Trading below 4H EMA50, reclaiming 1H EMA20 on volume/OI turn)
        w2_candidates = []
        for s, f in features_by_sym.items():
            if (
                f["close"] < f["ema50_4h"] * 0.985
                and f["close"] >= f["ema20_1h"]
                and f["ret_6h"] > 0.01
                and (f["vol_ratio_6h"] >= 1.35 or f["rvol_4h"] >= 1.35 or f["oi_6h"] >= 0.03)
            ):
                score = f["vol_ratio_6h"] * 20 + f["oi_6h"] * 100 + f["ret_6h"] * 100
                w2_candidates.append((score, s))
        w2_candidates.sort(reverse=True)

        # 5. Track 3 Watch W3: IMPULSE_CONTINUATION_FLAG_WATCH (Already broke out with bbw_4h > 0.22, holding above 4H EMA20 & 1H EMA20 with tight 6h consolidation or OI support)
        w3_candidates = []
        for s, f in features_by_sym.items():
            if (
                f["bbw_4h"] > 0.22
                and f["close"] >= f["ema20_4h"]
                and f["close"] >= f["ema20_1h"] * 0.99
                and f["ret_24h"] >= 0.06
                and -0.04 <= f["ret_6h"] <= 0.12
                and (f["oi_24h"] >= 0.05 or f["vol_ratio_6h"] >= 1.25)
            ):
                score = f["oi_24h"] * 80 + f["ret_24h"] * 50 + f["vol_ratio_6h"] * 15
                w3_candidates.append((score, s))
        w3_candidates.sort(reverse=True)

        def summarize_bucket(name: str, ranked: list[tuple[float, str]], budget: int = 12) -> dict[str, Any]:
            picked = [s for _, s in ranked[:budget]]
            if not picked:
                return {"name": name, "count": 0}
            hits = [s for s in picked if s in top10_set]
            rets = [forward_ret[s] for s in picked]
            mfes = [forward_mfe[s] for s in picked]
            maes = [forward_mae[s] for s in picked]
            win_5pct = sum(1 for m in mfes if m >= 0.05)
            win_10pct = sum(1 for m in mfes if m >= 0.10)
            pos_24h = sum(1 for r in rets if r > 0)
            return {
                "name": name,
                "total_qualified": len(ranked),
                "budget_selected": len(picked),
                "selected_symbols": picked,
                "top10_hits": len(hits),
                "top10_hit_symbols": hits,
                "mean_24h_ret_pct": round(sum(rets) / len(rets) * 100, 2),
                "mean_24h_mfe_pct": round(sum(mfes) / len(mfes) * 100, 2),
                "mean_24h_mae_pct": round(sum(maes) / len(maes) * 100, 2),
                "mfe_to_mae_ratio": round((sum(mfes) / len(mfes)) / max(abs(sum(maes) / len(maes)), 1e-8), 2),
                "pct_reaching_5pct_mfe": round(win_5pct / len(picked) * 100, 1),
                "pct_reaching_10pct_mfe": round(win_10pct / len(picked) * 100, 1),
                "pct_positive_24h_close": round(pos_24h / len(picked) * 100, 1),
            }

        # Combined Dual-System Desk (6 Coiled Challenger Entries + 2 W1 + 2 W2 + 2 W3 = 12 total budget)
        combined_desk: list[tuple[float, str]] = []
        seen: set[str] = set()
        for bucket, quota in [
            (challenger_core_candidates, 6),
            (w1_candidates, 2),
            (w2_candidates, 2),
            (w3_candidates, 2),
        ]:
            added = 0
            for sc, sym in bucket:
                if sym not in seen:
                    seen.add(sym)
                    combined_desk.append((sc, sym))
                    added += 1
                    if added >= quota:
                        break

        return {
            "window": label,
            "eligible_universe_count": len(universe_syms),
            "universe_baseline": {
                "mean_24h_ret_pct": round(u_mean_ret * 100, 2),
                "mean_24h_mfe_pct": round(u_mean_mfe * 100, 2),
                "mean_24h_mae_pct": round(u_mean_mae * 100, 2),
                "mfe_to_mae_ratio": round(u_mean_mfe / max(abs(u_mean_mae), 1e-8), 2),
            },
            "strategies_at_12_slot_budget": [
                summarize_bucket("1. Current Pre-Breakout Coiled Core (Stage 1 Un-enriched Proxy)", coiled_core_candidates, 12),
                summarize_bucket("2. Repaired Pre-Breakout Coiled Core (Track 2 Challenger)", challenger_core_candidates, 12),
                summarize_bucket("3. Watch W1: Stealth Flow & OI Divergence", w1_candidates, 12),
                summarize_bucket("4. Watch W2: Deep-Value / Bottom Reclaim", w2_candidates, 12),
                summarize_bucket("5. Watch W3: Post-Impulse High-Tight Flag", w3_candidates, 12),
                summarize_bucket("6. Combined Dual-Track System (6 Coiled Core + 6 Watch)", combined_desk, 12),
            ],
        }

    # Cutoff A: Nov 3, 2026 23:00 UTC (1793746800000 - 1) -> Nov 4 23:00 UTC
    # Wait, let's inspect exact close_time values in klines_by_sym["BTCUSDT"]
    btc_bars = klines_by_sym["BTCUSDT"]
    # Find bar closing near 2026-11-03 23:59:59 UTC and 2026-11-04 08:59:59 UTC
    nov3_23z_ms = max(b["close_time"] for b in btc_bars if datetime.fromtimestamp(b["close_time"] / 1000, tz=UTC) <= datetime(2026, 11, 4, 0, 0, tzinfo=UTC))
    nov4_23z_ms = max(b["close_time"] for b in btc_bars if datetime.fromtimestamp(b["close_time"] / 1000, tz=UTC) <= datetime(2026, 11, 5, 0, 0, tzinfo=UTC))

    nov4_09z_ms = max(b["close_time"] for b in btc_bars if datetime.fromtimestamp(b["close_time"] / 1000, tz=UTC) <= datetime(2026, 11, 4, 9, 0, tzinfo=UTC))
    nov5_09z_ms = max(b["close_time"] for b in btc_bars if datetime.fromtimestamp(b["close_time"] / 1000, tz=UTC) <= datetime(2026, 11, 5, 9, 30, tzinfo=UTC))

    win_a = evaluate_window(nov3_23z_ms, nov4_23z_ms, set(nov4_top10), "Window A: Nov 3 23:59Z -> Nov 4 23:59Z (24h Forward)")
    win_b = evaluate_window(nov4_09z_ms, nov5_09z_ms, set(nov5_top10), "Window B: Nov 4 08:59Z -> Nov 5 09:00Z (24h Forward)")

    return {"window_a": win_a, "window_b": win_b}


async def verify_track2_challenger_on_exact_screener() -> dict[str, Any]:

      - Control (Frozen v2.7 Core full pipeline)
